function internationalMasjidOwner(m) {
  let method = m.get('adminVerificationMethod', '');
  let password = request.auth != null && request.auth.token.firebase.sign_in_provider == 'password';
  return request.auth != null && m.get('adminUid', '') == request.auth.uid &&
    m.verificationStatus == 'Verified' &&
    request.auth.token.get('email', '') == m.adminEmail &&
    ((method == 'email_password' && password) ||
      (method == 'email' && request.auth.token.get('email_verified', false) == true &&
        (password || request.auth.token.get('adminEmailOtp', false) == true)));
}
function phoneMasjidOwner(m) {
  let mobile = m.adminMobile;
  let full = mobile.matches('^\\+.*') ? mobile : '+91' + mobile;
  return request.auth != null && m.get('adminVerificationMethod', 'phone') == 'phone' &&
    request.auth.token.get('phone_number', '') == full &&
    (m.get('adminUid', null) == null || m.adminUid == request.auth.uid);
}

function internationalOwnerSafeUpdate(before, after) {
  let changed = after.diff(before).affectedKeys();
  return internationalMasjidOwner(before) &&
    changed.hasOnly(['prayerTimes', 'timesUpdatedAt', 'timesConfirmedAt',
      'customAzanAudioName', 'customAzanAudioUrl', 'about', 'capacity', 'photoUrls']) &&
    (!changed.hasAny(['timesUpdatedAt']) || after.timesUpdatedAt == request.time) &&
    (!changed.hasAny(['timesConfirmedAt']) || after.timesConfirmedAt == request.time);
}

// Optional narrow permission for registration photos before approval. This is
// NOT permission to edit timings, ownership, contact details or status.
function pendingInternationalPhotoUpdate(before, after) {
  return request.auth != null && before.get('adminUid', '') == request.auth.uid &&
    before.get('adminVerificationMethod', '') == 'email_password' &&
    before.verificationStatus == 'Pending Verification' &&
    request.auth.token.firebase.sign_in_provider == 'password' &&
    request.auth.token.get('email', '') == before.adminEmail &&
    after.diff(before).affectedKeys().hasOnly(['photoUrls']) &&
    after.photoUrls is list && after.photoUrls.size() <= 3;
}
